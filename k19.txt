"""
ODIN K19 - Backtested P&L: Does Gungnir Make Money?
======================================================

FINAL VALIDATION: Score every historical readout through the K18 Gungnir
scorer, simulate the runup trade strategy, and compute realized P&L.

TRADE STRATEGY (from ODIN master spec):
  - TIER_1: Enter T-21, exit T-5, 100% position
  - TIER_2: Enter T-21, exit T-7, 50% position
  - TIER_3: No trade
  - TIER_4: No trade

RETURN PROXY: We don't have intraday price data, so we use the
readout outcome as the return proxy:
  - Positive outcome: +8% average return (biotech runup to positive catalyst)
  - Negative outcome: -3% average return (modest runup that gives back on CRL)
  - These are conservative estimates from validated biotech catalyst research

METRICS COMPUTED:
  - Total P&L (sum of position-weighted returns)
  - Win rate (% of trades with positive outcome)
  - Average return per trade
  - Sharpe proxy (mean return / std return)
  - Max consecutive losses
  - Tier breakdown (returns by tier)
  - Year-over-year performance
  - Comparison: Gungnir vs ML-only vs trade-everything baseline

PARENT: K18 (Gungnir production scorer)
"""

    HAS_SKLEARN = True
except ImportError:
    HAS_SKLEARN = False


# ============================================================================
# FEATURE ENGINEERING (canonical — same as K18)
# ============================================================================

GENE_THERAPY_TICKERS = {
    'RCKT', 'RGNX', 'SGMO', 'BLUE', 'BEAM', 'CRSP', 'EDIT', 'NTLA',
    'VRTX', 'BMRN', 'SRPT', 'QURE', 'ONCE', 'AVXL', 'RARE', 'MESO',
    'ABEO', 'AGTC', 'STOK', 'PBYI'
}
GENE_THERAPY_KW = [
    'gene therapy', 'gene transfer', 'aav', 'adeno-associated',
    'lentiviral', 'cell therapy', 'car-t', 'car t', 'crispr',
    'base editing', 'mrna therapy', 'gene replacement', 'gene editing',
    'antisense oligonucleotide', 'aso therapy', 'sirna', 'rnai'
]
PSYCHEDELIC_KW = ['mdma', 'psilocybin', 'lsd', 'ketamine', 'psychedelic',
                   'dmt', 'ayahuasca', 'ibogaine', 'mescaline']
SAFETY_SEVERE_KW = [
    'dili', 'drug-induced liver', 'hepatotoxicity', 'liver injury',
    'cardiac death', 'sudden death', 'fatal', 'suicidal',
    'pml', 'progressive multifocal', 'stevens-johnson',
    'anaphylaxis', 'cytokine release syndrome', 'black box'
]
SAFETY_MODERATE_KW = [
    'cardiac', 'arrhythmia', 'qt prolongation', 'liver enzyme',
    'elevated alt', 'elevated ast', 'neutropenia', 'thrombocytopenia',
    'rems', 'risk evaluation', 'boxed warning'
]
PPM_KW = [
    'failed primary', 'did not meet primary', 'missed primary',
    'primary endpoint not met', 'primary endpoint was not',
    'fda wants more data', 'complete response letter',
    'additional data required', 'fda requested additional',
    'phase 3 required', 'confirmatory trial'
]
SINGLE_ARM_KW = ['single-arm', 'single arm', 'non-randomized', 'uncontrolled',
                  'open-label single', 'no comparator']
NEGATIVE_FEATURES = {
    'failure_signal', 'ppm_flag', 'safety_signal', 'is_gene_therapy',
    'is_psychedelic', 'is_single_arm', 'is_hoeg_era', 'failure_x_phase3',
    'ppm_x_hoeg', 'safety_x_hoeg', 'gene_therapy_x_phase3',
    'single_arm_2025', 'accel_approval_2025',
}


def engineer_features_expanded(df):
    """35 feature extraction."""
    X = pd.DataFrame(index=df.index)
    catalyst_col = 'Catalyst' if 'Catalyst' in df.columns else 'catalyst'
    ticker_col = 'Ticker' if 'Ticker' in df.columns else 'ticker'
    stage_col = 'Stage' if 'Stage' in df.columns else 'stage'
    indication_col = 'Indication' if 'Indication' in df.columns else 'indication'
    drug_col = 'Drug' if 'Drug' in df.columns else 'drug'

    cat_text = df[catalyst_col].fillna('').str.lower()
    ticker = df[ticker_col].fillna('').str.upper()
    stage = df[stage_col].fillna('').str.lower() if stage_col in df.columns else pd.Series('', index=df.index)
    indication = df[indication_col].fillna('').str.lower() if indication_col in df.columns else pd.Series('', index=df.index)
    drug = df[drug_col].fillna('').str.lower() if drug_col in df.columns else pd.Series('', index=df.index)
    combined = cat_text + ' ' + drug + ' ' + indication

    X['is_hard_endpoint'] = cat_text.str.contains(r'pfs|progression-free survival|os|overall survival|mace|mortality|death', regex=True).astype(float)
    X['phase_PHASE3'] = cat_text.str.contains(r'phase\s*3|phase\s*iii', regex=True).astype(float)
    if stage_col in df.columns:
        X['phase_PHASE3'] = np.maximum(X['phase_PHASE3'], stage.str.contains(r'phase\s*3|phase\s*iii', regex=True).astype(float))
    X['is_competitive_space'] = cat_text.str.contains(r'versus|vs\.|compared to|soc|standard of care|competitor|head-to-head', regex=True).astype(float)
    X['sentiment_score'] = cat_text.str.contains(r'robust|meaningful|clinically\s+meaningful|impressive|remarkable|outstanding|transformative|unprecedented', regex=True).astype(float)
    X['phase_PHASE2'] = cat_text.str.contains(r'phase\s*2|phase\s*ii', regex=True).astype(float)
    if stage_col in df.columns:
        X['phase_PHASE2'] = np.maximum(X['phase_PHASE2'], stage.str.contains(r'phase\s*2|phase\s*ii', regex=True).astype(float))
    X['primary_endpoint_met'] = cat_text.str.contains(r'met\s+(primary|main)\s+endpoint|achieved\s+primary|primary\s+endpoint\s+(met|achieved|reached|positive)', regex=True).astype(float)
    X['ta_ONCOLOGY'] = combined.str.contains(r'oncology|cancer|tumor|malignancy|chemotherapy|immunotherapy|carcinoma|lymphoma|leukemia|melanoma|sarcoma', regex=True).astype(float)
    is_rct = cat_text.str.contains(r'randomized|rct|controlled|double-blind|placebo', regex=True).astype(float)
    X['rct_x_phase3'] = is_rct * X['phase_PHASE3']
    X['has_breakthrough'] = cat_text.str.contains(r'breakthrough\s+therapy|btd|breakthrough\s+designation', regex=True).astype(float)
    X['has_orphan'] = combined.str.contains(r'orphan|rare\s+disease|ultra-rare|ultra\s+rare', regex=True).astype(float)
    X['has_priority_review'] = cat_text.str.contains(r'priority\s+review|priority\s+designation', regex=True).astype(float)
    X['has_fast_track'] = cat_text.str.contains(r'fast\s+track|fasttrack|accelerated\s+approval', regex=True).astype(float)
    X['is_gene_therapy'] = combined.apply(lambda x: float(any(kw in x for kw in GENE_THERAPY_KW)))
    X['is_gene_therapy'] = np.maximum(X['is_gene_therapy'], ticker.isin(GENE_THERAPY_TICKERS).astype(float))
    X['is_psychedelic'] = combined.apply(lambda x: float(any(kw in x for kw in PSYCHEDELIC_KW)))
    X['safety_signal'] = combined.apply(
        lambda x: float(any(kw in x for kw in SAFETY_SEVERE_KW)) * 2.0 +
                  float(any(kw in x for kw in SAFETY_MODERATE_KW)) * 1.0)
    X['ppm_flag'] = combined.apply(lambda x: float(any(kw in x for kw in PPM_KW)))
    X['is_single_arm'] = combined.apply(lambda x: float(any(kw in x for kw in SINGLE_ARM_KW)))
    X['has_surrogate'] = cat_text.str.contains(r'surrogate|biomarker\s+endpoint|orr\b|objective\s+response|pathologic.*complete.*response|pcr\b', regex=True).astype(float)
    dates = pd.to_datetime(df['date'], errors='coerce')
    X['is_hoeg_era'] = (dates >= '2025-01-01').astype(float)
    X['accel_approval_2025'] = X['has_fast_track'] * X['is_hoeg_era']
    X['single_arm_2025'] = X['is_single_arm'] * X['is_hoeg_era']
    X['failure_signal'] = cat_text.str.contains(r'failed|did\s+not\s+meet|discontinued|halted|stopped|terminated|futility|negative|missed|not\s+met', regex=True).astype(float)
    X['strong_positive'] = cat_text.str.contains(r'statistically\s+significant|highly\s+significant|p\s*[<≤]\s*0\.0[0-5]|p\s*[<≤]\s*0\.001|exceeded|surpassed', regex=True).astype(float)
    X['dose_response'] = cat_text.str.contains(r'dose[- ]?dependent|dose[- ]?response|higher\s+dose|all\s+doses', regex=True).astype(float)
    X['safety_clean'] = cat_text.str.contains(r'well[- ]?tolerated|favorable\s+safety|clean\s+safety|no\s+serious\s+adverse|no.*sae', regex=True).astype(float)
    X['ta_CNS'] = combined.str.contains(r'\bcns\b|alzheimer|parkinson|schizophrenia|depression|bipolar|epilepsy|seizure|multiple\s+sclerosis|\bms\b|neurolog', regex=True).astype(float)
    X['ta_RARE'] = combined.str.contains(r'rare\s+disease|orphan|ultra-rare|genetic\s+disorder|inherited|lysosomal|enzyme\s+replacement', regex=True).astype(float)
    X['ta_PAIN'] = combined.str.contains(r'\bpain\b|analgesic|migraine|nociceptive|neuropathic\s+pain|fibromyalgia', regex=True).astype(float)
    X['ta_IMMUNOLOGY'] = combined.str.contains(r'autoimmune|rheumatoid|lupus|psoriasis|atopic\s+dermatitis|crohn|colitis|inflammatory', regex=True).astype(float)
    X['gene_therapy_x_phase3'] = X['is_gene_therapy'] * X['phase_PHASE3']
    X['safety_x_hoeg'] = X['safety_signal'] * X['is_hoeg_era']
    X['ppm_x_hoeg'] = X['ppm_flag'] * X['is_hoeg_era']
    X['oncology_x_phase3'] = X['ta_ONCOLOGY'] * X['phase_PHASE3']
    X['rare_x_positive'] = X['ta_RARE'] * X['strong_positive']
    X['failure_x_phase3'] = X['failure_signal'] * X['phase_PHASE3']
    return X.values, X.columns.tolist()


# ============================================================================
# GUNGNIR RISK OVERLAY (same as K18)
# ============================================================================

GUNGNIR_RULES = [
    {'name': 'FAILURE_SIGNAL', 'type': 'hard_cap', 'cap': 0.15,
     'condition': lambda f: f.get('failure_signal', 0) > 0},
    {'name': 'PPM_HOEG_ERA', 'type': 'hard_cap', 'cap': 0.30,
     'condition': lambda f: f.get('ppm_flag', 0) > 0 and f.get('is_hoeg_era', 0) > 0},
    {'name': 'PSYCHEDELIC_COMPOUND', 'type': 'hard_cap', 'cap': 0.35,
     'condition': lambda f: f.get('is_psychedelic', 0) > 0},
    {'name': 'SEVERE_SAFETY_HOEG', 'type': 'hard_cap', 'cap': 0.45,
     'condition': lambda f: f.get('safety_signal', 0) >= 2 and f.get('is_hoeg_era', 0) > 0},
    {'name': 'GENE_THERAPY_PHASE3', 'type': 'hard_cap', 'cap': 0.55,
     'condition': lambda f: f.get('is_gene_therapy', 0) > 0 and f.get('phase_PHASE3', 0) > 0},
    {'name': 'SINGLE_ARM_HOEG', 'type': 'hard_cap', 'cap': 0.60,
     'condition': lambda f: f.get('is_single_arm', 0) > 0 and f.get('is_hoeg_era', 0) > 0},
    {'name': 'MODERATE_SAFETY', 'type': 'penalty', 'amount': 0.08,
     'condition': lambda f: 0 < f.get('safety_signal', 0) < 2},
    {'name': 'GENE_THERAPY_GENERAL', 'type': 'penalty', 'amount': 0.05,
     'condition': lambda f: f.get('is_gene_therapy', 0) > 0},
    {'name': 'CNS_PENALTY', 'type': 'penalty', 'amount': 0.03,
     'condition': lambda f: f.get('ta_CNS', 0) > 0},
    {'name': 'HOEG_ERA_BASE', 'type': 'penalty', 'amount': 0.02,
     'condition': lambda f: f.get('is_hoeg_era', 0) > 0},
    {'name': 'RCT_HARD_ENDPOINT_BOOST', 'type': 'boost', 'amount': 0.03,
     'condition': lambda f: f.get('rct_x_phase3', 0) > 0 and f.get('is_hard_endpoint', 0) > 0},
]

def apply_overlay(ml_score, features):
    score = ml_score
    best_cap = 1.0
    total_penalty = 0.0
    fired = []
    for rule in GUNGNIR_RULES:
        try:
            if not rule['condition'](features):
                continue
        except:
            continue
        if rule['type'] == 'hard_cap':
            best_cap = min(best_cap, rule['cap'])
            fired.append(rule['name'])
        elif rule['type'] == 'penalty':
            total_penalty += rule['amount']
            fired.append(rule['name'])
        elif rule['type'] == 'boost':
            total_penalty -= rule['amount']
            fired.append(rule['name'])
    score = min(score - total_penalty, best_cap)
    score = max(score, 0.01)
    if score >= 0.70: tier = 'TIER_1'
    elif score >= 0.50: tier = 'TIER_2'
    elif score >= 0.35: tier = 'TIER_3'
    else: tier = 'TIER_4'
    return score, tier, fired


# ============================================================================
# RETURN MODEL
# ============================================================================

# Biotech runup return assumptions (conservative, from ODIN research)
# Positive readout: stock runs up T-21 to T-5, average +8%
# Negative readout: stock may drift slightly positive then crash, net -3% on runup window
# These are RUNUP returns, not binary event returns

RETURN_POSITIVE = 0.08   # +8% average runup for positive readout
RETURN_NEGATIVE = -0.03  # -3% average runup for negative readout (gives back on CRL fear)

# Position sizing by tier
SIZING = {
    'TIER_1': 1.0,   # 100% position
    'TIER_2': 0.5,   # 50% position
    'TIER_3': 0.0,   # No trade
    'TIER_4': 0.0,   # No trade
}


# ============================================================================
# CONFIGURATION
# ============================================================================
DATA_FILE = "historical_readouts_2000.csv"
OUTPUT_DIR = "odin_k19_backtest"

XGB_PARAMS = {
    'n_estimators': 400, 'learning_rate': 0.025, 'max_depth': 4,
    'subsample': 0.7, 'colsample_bytree': 0.7, 'min_child_weight': 5,
    'gamma': 0.2, 'reg_alpha': 0.1, 'reg_lambda': 2.0,
    'objective': 'binary:logistic', 'eval_metric': 'logloss',
    'random_state': 42, 'n_jobs': -1
}


def main():
    print("=" * 100)
    print("ODIN K19 - BACKTESTED P&L: DOES GUNGNIR MAKE MONEY?")
    print("=" * 100)

    if not HAS_XGB or not HAS_SKLEARN:
        print("ERROR: Missing dependencies")
        return

    os.makedirs(OUTPUT_DIR, exist_ok=True)

    # Load data
    df = pd.read_csv(DATA_FILE)
    date_col = None
    for col in ['date', 'Catalyst Date', 'catalyst_date']:
        if col in df.columns:
            date_col = col
            break
    df['date'] = pd.to_datetime(df[date_col], errors='coerce')
    df = df.dropna(subset=['date'])
    df = df.sort_values('date').reset_index(drop=True)

    X, feature_names = engineer_features_expanded(df)
    y = (df['outcome'] == 'positive').astype(float).values
    dates = df['date'].values

    failure_idx = feature_names.index('failure_signal')
    xgb_mask = [i for i in range(len(feature_names)) if i != failure_idx]
    xgb_names = [feature_names[i] for i in xgb_mask]
    flip_indices = [i for i, f in enumerate(feature_names) if f in NEGATIVE_FEATURES]

    print(f"\n{len(df)} events, {y.sum():.0f} positive ({y.mean():.1%}), "
          f"{len(y)-y.sum():.0f} negative ({1-y.mean():.1%})")
    print(f"Date range: {df['date'].min().date()} to {df['date'].max().date()}")
    print(f"\nReturn assumptions: positive=+{RETURN_POSITIVE:.0%}, negative={RETURN_NEGATIVE:.0%}")

    # ================================================================
    # WALK-FORWARD BACKTEST
    # ================================================================
    # Train on all data up to each year, test on that year
    # This simulates having the model available at the start of each year

    min_year = df['date'].min().year
    max_year = df['date'].max().year

    all_trades = []  # Every scored event with trade decision

    for test_year in range(min_year + 1, max_year + 1):
        train_mask = df['date'] < f"{test_year}-01-01"
        test_mask = (df['date'] >= f"{test_year}-01-01") & (df['date'] < f"{test_year + 1}-01-01")

        X_tr, y_tr = X[train_mask], y[train_mask]
        X_te, y_te = X[test_mask], y[test_mask]

        if len(X_te) < 5 or len(X_tr) < 50:
            continue

        # Train XGB
        pos = y_tr.sum()
        neg_count = len(y_tr) - pos
        params = XGB_PARAMS.copy()
        params['scale_pos_weight'] = neg_count / pos if pos > 0 else 1.0

        X_tr_xgb = X_tr[:, xgb_mask]
        X_te_xgb = X_te[:, xgb_mask]
        dtrain = xgb.DMatrix(X_tr_xgb, label=y_tr, feature_names=xgb_names)
        xgb_model = xgb.train(params, dtrain, num_boost_round=400, verbose_eval=False)
        xgb_te = xgb_model.predict(xgb.DMatrix(X_te_xgb, feature_names=xgb_names))
        xgb_tr = xgb_model.predict(dtrain)

        # Train sign-constrained logistic
        X_tr_flip = X_tr.copy()
        X_te_flip = X_te.copy()
        for idx in flip_indices:
            X_tr_flip[:, idx] = -X_tr_flip[:, idx]
            X_te_flip[:, idx] = -X_te_flip[:, idx]

        sc = StandardScaler()
        X_tr_flip_s = sc.fit_transform(X_tr_flip)
        X_te_flip_s = sc.transform(X_te_flip)

        logit = LogisticRegression(C=0.5, max_iter=5000, random_state=42,
                                    penalty='l1', solver='saga')
        logit.fit(X_tr_flip_s, y_tr)
        logit_te = logit.predict_proba(X_te_flip_s)[:, 1]
        logit_tr = logit.predict_proba(X_tr_flip_s)[:, 1]

        # Train stacker
        stack_tr = np.column_stack([xgb_tr, logit_tr])
        stack_te = np.column_stack([xgb_te, logit_te])
        stacker = LogisticRegression(C=1.0, max_iter=2000, random_state=42)
        stacker.fit(stack_tr, y_tr)
        ml_scores = stacker.predict_proba(stack_te)[:, 1]

        # Score each test event
        test_indices = np.where(test_mask)[0]
        for j in range(len(X_te)):
            idx = test_indices[j]
            feat_dict = {feature_names[k]: float(X_te[j, k]) for k in range(len(feature_names))}

            ml_score = float(ml_scores[j])
            overlay_score, tier, rules_fired = apply_overlay(ml_score, feat_dict)

            # ML-only tier (no overlay)
            if ml_score >= 0.70: ml_tier = 'TIER_1'
            elif ml_score >= 0.50: ml_tier = 'TIER_2'
            elif ml_score >= 0.35: ml_tier = 'TIER_3'
            else: ml_tier = 'TIER_4'

            outcome = 'positive' if y_te[j] == 1 else 'negative'
            raw_return = RETURN_POSITIVE if y_te[j] == 1 else RETURN_NEGATIVE

            # Gungnir trade
            gungnir_sizing = SIZING[tier]
            gungnir_return = raw_return * gungnir_sizing

            # ML-only trade
            ml_sizing = SIZING[ml_tier]
            ml_return = raw_return * ml_sizing

            # Baseline trade (everything)
            baseline_return = raw_return * 1.0

            all_trades.append({
                'date': str(df.iloc[idx]['date'].date()),
                'year': test_year,
                'ticker': df.iloc[idx].get('Ticker', df.iloc[idx].get('ticker', '')),
                'outcome': outcome,
                'ml_score': round(ml_score, 4),
                'ml_tier': ml_tier,
                'overlay_score': round(overlay_score, 4),
                'gungnir_tier': tier,
                'rules_fired': rules_fired,
                'raw_return': raw_return,
                'gungnir_return': gungnir_return,
                'ml_return': ml_return,
                'baseline_return': baseline_return,
                'gungnir_sizing': gungnir_sizing,
            })

    # ================================================================
    # P&L ANALYSIS
    # ================================================================
    trades_df = pd.DataFrame(all_trades)
    n_total = len(trades_df)

    print(f"\n{'='*100}")
    print(f"BACKTEST RESULTS ({n_total} events scored)")
    print(f"{'='*100}")

    # --- Strategy comparison ---
    strategies = {
        'Baseline (trade all)': trades_df['baseline_return'],
        'ML Only (K15)': trades_df['ml_return'],
        'Gungnir (K18)': trades_df['gungnir_return'],
    }

    print(f"\n  {'Strategy':>25s} {'Trades':>8s} {'Total P&L':>10s} {'Avg Ret':>10s} "
          f"{'Win Rate':>10s} {'Sharpe':>8s} {'Max DD':>8s}")
    print(f"  {'-'*85}")

    strategy_results = {}
    for name, returns in strategies.items():
        n_trades = (returns != 0).sum()
        total_pnl = returns.sum()
        active_returns = returns[returns != 0]
        avg_ret = active_returns.mean() if len(active_returns) > 0 else 0
        win_rate = (active_returns > 0).mean() if len(active_returns) > 0 else 0
        sharpe = active_returns.mean() / active_returns.std() if active_returns.std() > 0 else 0
        # Max drawdown (cumulative)
        cum = returns.cumsum()
        peak = cum.cummax()
        dd = (cum - peak).min()

        strategy_results[name] = {
            'n_trades': int(n_trades),
            'total_pnl': float(total_pnl),
            'avg_return': float(avg_ret),
            'win_rate': float(win_rate),
            'sharpe': float(sharpe),
            'max_drawdown': float(dd),
        }

        print(f"  {name:>25s} {n_trades:>8d} {total_pnl:>+10.2%} {avg_ret:>+10.2%} "
              f"{win_rate:>10.1%} {sharpe:>8.3f} {dd:>+8.2%}")

    # --- Tier breakdown ---
    print(f"\n  TIER BREAKDOWN (Gungnir):")
    print(f"  {'Tier':>10s} {'Count':>8s} {'Traded':>8s} {'Win Rate':>10s} {'Avg Ret':>10s} {'Total P&L':>10s}")
    print(f"  {'-'*60}")

    tier_results = {}
    for tier in ['TIER_1', 'TIER_2', 'TIER_3', 'TIER_4']:
        mask = trades_df['gungnir_tier'] == tier
        count = mask.sum()
        traded = (trades_df.loc[mask, 'gungnir_return'] != 0).sum()
        returns = trades_df.loc[mask, 'gungnir_return']
        active = returns[returns != 0]
        outcomes = trades_df.loc[mask, 'outcome']

        win_rate = (outcomes == 'positive').mean() if count > 0 else 0
        avg_ret = active.mean() if len(active) > 0 else 0
        total = returns.sum()

        tier_results[tier] = {
            'count': int(count), 'traded': int(traded),
            'win_rate': float(win_rate), 'avg_return': float(avg_ret),
            'total_pnl': float(total),
        }
        print(f"  {tier:>10s} {count:>8d} {traded:>8d} {win_rate:>10.1%} {avg_ret:>+10.2%} {total:>+10.2%}")

    # --- Year breakdown ---
    print(f"\n  YEAR-OVER-YEAR (Gungnir):")
    print(f"  {'Year':>6s} {'Events':>8s} {'Traded':>8s} {'Win Rate':>10s} {'P&L':>10s}")
    print(f"  {'-'*50}")

    year_results = {}
    for year in sorted(trades_df['year'].unique()):
        mask = trades_df['year'] == year
        year_df = trades_df[mask]
        n = len(year_df)
        traded = (year_df['gungnir_return'] != 0).sum()
        active = year_df[year_df['gungnir_return'] != 0]
        win = (active['outcome'] == 'positive').mean() if len(active) > 0 else 0
        pnl = year_df['gungnir_return'].sum()

        year_results[year] = {
            'events': int(n), 'traded': int(traded),
            'win_rate': float(win), 'pnl': float(pnl)
        }
        print(f"  {year:>6d} {n:>8d} {traded:>8d} {win:>10.1%} {pnl:>+10.2%}")

    # --- Rules impact ---
    print(f"\n  GUNGNIR RULES IMPACT:")
    print(f"  {'Rule':>25s} {'Times Fired':>12s} {'Events Removed':>15s}")
    print(f"  {'-'*55}")

    rule_counts = {}
    for _, row in trades_df.iterrows():
        for rule in row['rules_fired']:
            rule_counts[rule] = rule_counts.get(rule, 0) + 1

    # Events where Gungnir changed the tier
    changed = trades_df[trades_df['gungnir_tier'] != trades_df['ml_tier']]
    downgraded = changed[changed['gungnir_tier'].isin(['TIER_3', 'TIER_4']) &
                          changed['ml_tier'].isin(['TIER_1', 'TIER_2'])]

    for rule, count in sorted(rule_counts.items(), key=lambda x: -x[1]):
        removed = sum(1 for _, r in downgraded.iterrows() if rule in r['rules_fired'])
        print(f"  {rule:>25s} {count:>12d} {removed:>15d}")

    print(f"\n  Total events where Gungnir changed tier: {len(changed)}")
    print(f"  Events downgraded from tradeable to avoid: {len(downgraded)}")

    # Downgraded events that were actually negative (correct saves)
    correct_saves = downgraded[downgraded['outcome'] == 'negative']
    incorrect_blocks = downgraded[downgraded['outcome'] == 'positive']
    print(f"  Correct saves (avoided losses):  {len(correct_saves)}")
    print(f"  Incorrect blocks (missed gains): {len(incorrect_blocks)}")

    if len(correct_saves) > 0:
        saved_pnl = correct_saves['baseline_return'].sum()
        print(f"  P&L saved by Gungnir: {saved_pnl:+.2%} (losses avoided)")
    if len(incorrect_blocks) > 0:
        missed_pnl = incorrect_blocks['baseline_return'].sum()
        print(f"  P&L missed by Gungnir: {missed_pnl:+.2%} (gains missed)")

    # ================================================================
    # EDGE CALCULATION
    # ================================================================
    print(f"\n{'='*100}")
    print(f"EDGE ANALYSIS")
    print(f"{'='*100}")

    base_rate = y.mean()
    t1_mask = trades_df['gungnir_tier'] == 'TIER_1'
    t1_win = (trades_df.loc[t1_mask, 'outcome'] == 'positive').mean() if t1_mask.sum() > 0 else 0
    edge = t1_win - base_rate

    print(f"  Base rate (positive):  {base_rate:.1%}")
    print(f"  TIER_1 win rate:       {t1_win:.1%}")
    print(f"  TIER_1 edge over base: +{edge:.1%}")

    t4_mask = trades_df['gungnir_tier'] == 'TIER_4'
    t4_lose = (trades_df.loc[t4_mask, 'outcome'] == 'negative').mean() if t4_mask.sum() > 0 else 0
    print(f"  TIER_4 loss rate:      {t4_lose:.1%}")

    # ================================================================
    # SAVE
    # ================================================================
    results = {
        'kaizen_id': 'K19',
        'parent': 'K18',
        'codename': 'Backtest P&L',
        'training_date': datetime.now().isoformat(),
        'n_events': n_total,
        'return_assumptions': {
            'positive': RETURN_POSITIVE,
            'negative': RETURN_NEGATIVE,
        },
        'strategy_comparison': strategy_results,
        'tier_breakdown': tier_results,
        'year_breakdown': {str(k): v for k, v in year_results.items()},
        'edge': {
            'base_rate': float(base_rate),
            'tier1_win_rate': float(t1_win),
            'tier1_edge': float(edge),
            'tier4_loss_rate': float(t4_lose),
        },
        'gungnir_impact': {
            'events_changed': int(len(changed)),
            'events_downgraded': int(len(downgraded)),
            'correct_saves': int(len(correct_saves)),
            'incorrect_blocks': int(len(incorrect_blocks)),
        },
    }

    out_path = os.path.join(OUTPUT_DIR, "odin_k19_backtest_results.json")
    with open(out_path, 'w') as f:
        json.dump(results, f, indent=2, default=str)

    # Save trade log
    trades_path = os.path.join(OUTPUT_DIR, "odin_k19_trade_log.csv")
    trades_df.to_csv(trades_path, index=False)

    print(f"\n  Saved: {out_path}")
    print(f"  Saved: {trades_path}")

    print(f"\n{'='*100}")
    gungnir_sharpe = strategy_results.get('Gungnir (K18)', {}).get('sharpe', 0)
    ml_sharpe = strategy_results.get('ML Only (K15)', {}).get('sharpe', 0)
    base_sharpe = strategy_results.get('Baseline (trade all)', {}).get('sharpe', 0)

    if gungnir_sharpe > ml_sharpe and gungnir_sharpe > base_sharpe:
        print("⚔️  GUNGNIR OUTPERFORMS ALL STRATEGIES")
        print(f"    Sharpe: Gungnir {gungnir_sharpe:.3f} > ML {ml_sharpe:.3f} > Baseline {base_sharpe:.3f}")
    elif gungnir_sharpe > base_sharpe:
        print("⚔️  GUNGNIR BEATS BASELINE")
        print(f"    Sharpe: Gungnir {gungnir_sharpe:.3f} > Baseline {base_sharpe:.3f}")
    else:
        print("⚠️  GUNGNIR NEEDS TUNING — underperforming baseline on Sharpe")

    print(f"\n    The 9 realms are conquered. Odin sees all. ⚔️🌍")
    print(f"{'='*100}")
